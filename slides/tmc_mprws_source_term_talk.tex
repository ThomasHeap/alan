% =============================================================
\section{The problem}
% =============================================================

\begin{frame}{Why one sample per latent isn't enough}
  \begin{itemize}
    \item Standard importance sampling / VI: draw \emph{one} sample of the
      whole latent vector $z$ from $Q$, reweight by $p(z,x)/q(z)$.
    \item Better: draw $K$ \emph{whole-vector} samples (IWAE-style) and
      average: tighter bound, lower variance, but cost is $O(K)$ per
      \emph{full model}, and you need $K$ large before it helps on models
      with many latents.
    \item \alert{What if we could afford $K$ samples \emph{per latent
      variable}, not per whole vector?} Naively that's $K^{\#\text{latents}}$
      combinations to consider: looks hopeless.
  \end{itemize}
  \vfill
  This is the problem Tensor Monte Carlo, then Massively-Parallel RWS, and
  then the source term trick, solve in turn.
\end{frame}

\begin{frame}{Wake-Sleep, briefly}
  Classic wake-sleep (Hinton et al.) trains a generative model $P$ and a
  recognition/proposal model $Q$ together:
  \begin{center}
  \begin{tikzpicture}[node distance=14mm,
      box/.style={draw,rounded corners,minimum width=32mm,minimum height=9mm,align=center,font=\small}]
    \node[box,fill=talkteal!15] (wake) {\textbf{Wake phase}\\ fit $P$ so $Q$'s samples\\ look like the data};
    \node[box,fill=talkcoral!15,right=28mm of wake] (sleep) {\textbf{Sleep phase}\\ fit $Q$ to invert $P$\\ (recognize $P$'s samples)};
    \draw[-{Latex[length=2mm]}] ([yshift=2mm]wake.east) -- ([yshift=2mm]sleep.west);
    \draw[-{Latex[length=2mm]}] ([yshift=-2mm]sleep.west) -- ([yshift=-2mm]wake.east);
  \end{tikzpicture}
  \end{center}
  \small\textbf{Reweighted Wake-Sleep (RWS)} improves the $Q$ update: instead
  of training $Q$ on $P$'s own samples, use self-normalized importance
  weights from $K$ samples of $Q$ to push $Q$ toward the \emph{true
  posterior} directly. Unlike VI, RWS never reparameterizes: it works
  directly with \alert{discrete latent variables}, no REINFORCE or
  relaxations needed.
\end{frame}

% =============================================================
\section{Tensor Monte Carlo}
% =============================================================

\begin{frame}{Tensor Monte Carlo (Aitchison, 2019)}
  The key move that makes ``$K$ samples per latent'' tractable:
  \begin{itemize}
    \item Draw $K$ candidate values for \emph{each} latent variable
      separately, not $K$ whole-vector samples.
    \item Most latents only interact with a handful of neighbours in the
      model's dependency graph (plates, conditional independence).
    \item So the $K^{\#\text{latents}}$ sum factorizes into a sequence of
      small tensor contractions (like \texttt{einsum}) that follow the
      graph structure: cost becomes polynomial in $K$, not exponential.
  \end{itemize}
\end{frame}

\begin{frame}{Why this matters: a 4-latent chain, $K=3$}
  \begin{center}
  \begin{tikzpicture}[scale=0.62,
      node distance=16mm,
      lat/.style={circle,draw,minimum size=7mm,font=\scriptsize}]
    \node[lat] (a) {A};
    \node[lat,right=of a] (b) {B};
    \node[lat,right=of b] (c) {C};
    \node[lat,right=of c] (d) {D};
    \draw[-{Latex[length=1.8mm]}] (a) -- (b);
    \draw[-{Latex[length=1.8mm]}] (b) -- (c);
    \draw[-{Latex[length=1.8mm]}] (c) -- (d);
  \end{tikzpicture}
  \end{center}
  \small
  Each factor only touches its neighbour, so the sum \alert{distributes}: pull
  each inner sum out past every factor that doesn't depend on it, one latent
  at a time.
  \[
    \hat P_{\text{MP}} = \sum_{k_a,k_b,k_c,k_d} f_A(k_a)\, f_{AB}(k_a,k_b)\, f_{BC}(k_b,k_c)\, f_{CD}(k_c,k_d)
  \]
  \vspace{-2mm}
  \begin{align*}
    &= \sum_{k_b,k_c,k_d} v_B[k_b]\, f_{BC}(k_b,k_c)\, f_{CD}(k_c,k_d),
      &&v_B[k_b] = \sum_{k_a} f_A(k_a)\, f_{AB}(k_a,k_b) \\
    &= \sum_{k_c,k_d} v_C[k_c]\, f_{CD}(k_c,k_d),
      &&v_C[k_c] = \sum_{k_b} v_B[k_b]\, f_{BC}(k_b,k_c) \\
    &= \sum_{k_d} v_D[k_d],
      &&v_D[k_d] = \sum_{k_c} v_C[k_c]\, f_{CD}(k_c,k_d)
  \end{align*}
  \vspace{-2mm}
  \small three $K{\times}K$ contractions, $3K^2 = 27$ multiply-adds, not
  $K^4 = 81$: the saving grows with every extra latent in the chain.
\end{frame}

\begin{frame}{What TMC gives you (and doesn't)}
  \begin{itemize}
    \item[\alert{+}] A much lower-variance, $K$-per-latent marginal
      likelihood estimator, at polynomial cost.
    \item[\alert{--}] On its own, it's just an estimator: no prescription
      for \emph{training} $P$ and $Q$ together, and no cheap way to get
      posterior samples/moments back out afterward.
  \end{itemize}
  \vfill
  \textbf{Massively-Parallel Reweighted Wake-Sleep} (Heap et al., UAI 2023)
  turns this estimator into a full training algorithm.
\end{frame}

% =============================================================
\section{Massively-Parallel RWS}
% =============================================================

\begin{frame}{The estimator: $P_{\text{MP}}(z)$}
  For each latent, draw $K$ particles from $Q$; contract following the plate
  structure (as above) to get a single scalar:
  \[
    \hat{P}_{\text{MP}}(x) \;=\; \text{(TMC-style contraction of } p(z,x)/q(z)
    \text{ over all latents' } K \text{ particles)}
  \]
  \begin{itemize}
    \item \alert{Unbiased}: $\mathbb{E}[\hat P_{\text{MP}}(x)] = p(x)$, the
      true marginal likelihood, proved directly, not just assumed.
    \item This single fact is what later makes \emph{exact} MCMC schemes
      (Particle MCMC) possible on top of this machinery; more on that at
      the end.
  \end{itemize}
\end{frame}

\begin{frame}{Two updates, one estimator}
  Practical form of the massively-parallel RWS updates (Heap et al., Eq.\ 19):
  \[
    \Delta\theta_{\text{MP}} = \mathbb{E}_{Q_{\text{MP}}(z|x)}\big[\nabla_\theta \log \hat P_{\text{MP}}(z)\big]
    \qquad
    \Delta\phi_{\text{MP}} = \mathbb{E}_{Q_{\text{MP}}(z|x)}\big[\nabla_\phi \big(-\log \hat P_{\text{MP}}(z)\big)\big]
  \]
  \begin{itemize}
    \item \textbf{Wake phase}: ascend $\Delta\theta_{\text{MP}}$, i.e.\ climb
      $\log \hat P_{\text{MP}}$ w.r.t.\ $P$'s parameters.
    \item \textbf{Sleep phase}: ascend $\Delta\phi_{\text{MP}}$, i.e.\
      \emph{descend} $\log \hat P_{\text{MP}}$ w.r.t.\ $Q$'s parameters, which
      pushes $Q$ toward the true posterior.
  \end{itemize}
  \vfill
  \small Equivalent to the fuller self-normalized-importance-weight form
  (Eq.\ 18): a weighted average of $\nabla \log P_\theta(z^{\mathbf{k}},x)$ /
  $\nabla \log Q_\phi(z^{\mathbf{k}},x)$ over all $K^n$ combinations, weighted
  by $r_{\mathbf{k}}(z)/\hat P_{\text{MP}}(z)$. Both forms reuse the same
  $\hat P_{\text{MP}}(z)$ computation; no separate machinery for $P$ vs.\ $Q$.
\end{frame}

\begin{frame}{Results: MP RWS vs.\ global RWS, MovieLens}
  \begin{center}
    \includegraphics[width=0.6\textwidth]{figures/mprws_movielens_results.png}
  \end{center}
  \small Predictive log-likelihood vs.\ $K \in \{3,10,30\}$, at three dataset
  sizes. MP RWS (red) beats global RWS (blue) at every $K$ and every size,
  and the gap widens as the number of users grows (Heap et al., 2023).
\end{frame}

% =============================================================
\section{MPIW}
% =============================================================

\begin{frame}{From ``global'' to massively-parallel importance weighting}
  MP-RWS trains $P$ and $Q$. \textbf{MPIW} (Bowyer, Heap \& Aitchison, UAI
  2024) asks a different question: once trained, how do we get
  \emph{posterior expectations} back out?
  \begin{itemize}
    \item \textbf{Global} importance weighting: draw $K$ whole-vector
      samples, reweight, average: the classical scheme.
    \item \textbf{Massively-parallel} importance weighting (MPIW): draw $K$
      \emph{per latent}, reweight \emph{all} $K^{\#\text{latents}}$
      combinations (same trick as before), and use \emph{that} as the
      importance-sampling estimator of a posterior expectation.
  \end{itemize}
  \vfill
  \small This is the same $\hat P_{\text{MP}}(z)$ construction as before, now
  reframed as a general-purpose importance-weighting tool in its own right,
  not just something living inside the RWS training loop.
\end{frame}

\begin{frame}{The math}
  With $\mathbf{k}=(k_1,\dots,k_n)$ indexing one combination of particles
  (one per latent), and $\text{qa}(i)$ the parents of latent $i$ under $Q$:
  \[
    r_{\mathbf{k}}(z) = \frac{P(x,z^{\mathbf{k}})}{\prod_i Q_{\text{MP}}\big(z_i^{k_i} \mid z_j \text{ for } j\in\text{qa}(i)\big)}
    \qquad
    \hat P_{\text{MP}}(z) = \frac{1}{K^n}\sum_{\mathbf{k}\in\mathcal{K}^n} r_{\mathbf{k}}(z)
  \]
  \vfill
  \small $r_{\mathbf{k}}(z)$ factorizes into low-rank tensors along the graph
  (as on the TMC slide), so this sum is computed via \texttt{opt\_einsum},
  never by looping over all $K^n$ terms directly, giving one scalar,
  $\hat P_{\text{MP}}(z)$, per forward pass.
\end{frame}

\section{The source term trick}
% =============================================================

\begin{frame}{Why can't we just calculate moments like that?}
  A posterior moment estimate looks like a similarly simple sum:
  \[
    \hat m_{\text{MP}}(z) = \frac{1}{K^n}\sum_{\mathbf{k}\in\mathcal{K}^n} \frac{r_{\mathbf{k}}(z)}{\hat P_{\text{MP}}(z)}\, m(z^{\mathbf{k}})
  \]
  \begin{itemize}
    \item $\hat P_{\text{MP}}(z)$ is one \emph{forward} pass along the graph
      (contract, get a scalar): the TMC/MPIW trick handles that.
    \item But $\hat m_{\text{MP}}(z)$ needs, for \emph{each individual
      particle}, how much it contributed to that total. That means a
      \alert{backward} pass through the same contraction, like backprop
      through the sum-product graph.
    \item That backward pass is genuinely complex, and \alert{different for
      every quantity}: a sample, a marginal, a mean, a covariance each need
      their own hand-written traversal, for every plate topology.
  \end{itemize}
\end{frame}

\begin{frame}{The math of the source term trick}
  Add a source term $J$ into the log joint, forming a modified normalizing
  constant (Bowyer, Heap \& Aitchison, UAI 2024):
  \[
    Z_m(J) = \int \! dz'\, P(x,z')\, e^{J\,m(z')}
    \qquad\Rightarrow\qquad
    \left.\frac{\partial}{\partial J}\log Z_m(J)\right|_{J=0} = \mathbb{E}_{P(z'|x)}[m(z')]
  \]
  The massively-parallel analogue replaces the integral with the same
  contraction as before, now carrying $J$ through it:
  \[
    \hat P_{\text{MP}}^{\exp}(z,J) = \frac{1}{K^n}\sum_{\mathbf{k}} r_{\mathbf{k}}(z)\, e^{J\,m(z^{\mathbf{k}})}
    \qquad\Rightarrow\qquad
    \left.\frac{\partial}{\partial J}\log \hat P_{\text{MP}}^{\exp}(z,J)\right|_{J=0} = \hat m_{\text{MP}}(z)
  \]
  \vfill
  \small $J=0$ recovers $\hat P_{\text{MP}}(z)$ exactly, so this costs one
  extra forward pass plus one ordinary autodiff call, not a hand-written
  backward traversal.
\end{frame}

\begin{frame}{Why this is the elegant part}
  \begin{itemize}
    \item No new algorithm per quantity: \texttt{alan} computes
      posterior samples, marginals, \emph{and} moments through literally the
      same code path that computes the bound.
    \item Works uniformly across arbitrarily nested plates: the
      contraction already handles that; the source term rides along for
      free.
    \item It's autodiff doing the bookkeeping that would otherwise be a
      hand-written, model-specific resampling routine.
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{This is really in the code}
  From \texttt{alan}'s \texttt{Sample.py} (lightly trimmed):
  \begin{center}
  \begin{tikzpicture}[node distance=3mm,
      box/.style={draw,rounded corners,align=left,font=\footnotesize\ttfamily,minimum width=105mm,inner sep=3mm}]
    \node[box,fill=talkteal!10] (a) {dims = [*Kdims, *active\_platedims]\ \ \ \# same dims as the target latent(s)\\
      J\_tensor = zeros([d.size for d in dims], requires\_grad=True)\\
      L = self.\_elbo(sample, extra\_log\_factors=\{target: J\_tensor[dims]\})\\
      result = grad(L, J\_tensor)\ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \ \# same shape as J\_tensor};
  \end{tikzpicture}
  \end{center}
  \vfill
  \begin{itemize}
    \item \texttt{J\_tensor} starts at all zeros: it doesn't change what
      \texttt{L} evaluates to, but it has to carry the \emph{same} K-dim(s)
      (and plate dims) as whatever you're differentiating for, so the
      gradient comes back the same shape: one number per particle.
    \item One \texttt{grad} call replaces a bespoke backward traversal;
      samples, marginals, and moments all use this exact shape, only which
      dims \texttt{J} carries and what's baked into $L$ changes.
  \end{itemize}
\end{frame}

\section{QEM}
% =============================================================

\begin{frame}{Training $Q$ still needs gradients}
  MP-VI and MP-RWS both update $Q$'s parameters $\phi$ by \textbf{gradient
  ascent} on a massively-parallel bound.
  \begin{itemize}
    \item Needs a learning rate, tuned per model.
    \item \alert{Not invariant to reparameterization}: rescale a latent by a
      constant (leaving the actual distribution over data untouched) and
      gradient descent's behaviour changes, and sometimes it stops converging
      at all.
  \end{itemize}
  \vfill
  \textbf{QEM} (Heap, Bowyer \& Aitchison, 2025) removes gradient descent
  from the $Q$ update entirely.
\end{frame}

\begin{frame}{The QEM idea}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{E-step}
        Use MPIW + the source term trick to compute the \emph{current}
        posterior moments, exactly as before.
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{M-step}
        Set $Q$'s \emph{next} parameters directly from those moments, in
        closed form, for exponential-family $Q$ (Gaussian, Gamma, Beta,
        Dirichlet, Binomial, \dots).
      \end{block}
    \end{column}
  \end{columns}
  \vfill
  \small The name: it resembles EM, but optimizes the parameters of the
  \emph{approximate posterior} $Q$, not the model's prior/likelihood
  parameters $\theta$, as classical EM does. Hence ``QEM.''
\end{frame}

\begin{frame}[fragile]{Algorithm}
  \begin{center}
  \begin{tikzpicture}[node distance=2mm,
      box/.style={draw,rounded corners,align=left,font=\footnotesize\ttfamily,minimum width=105mm,inner sep=3mm}]
    \node[box,fill=talkteal!10] (a) {for t in 1..T:\\
      \ \ Q\_t = set\_exp\_family\_params(m[t-1])\ \ \# M-step\\
      \ \ z = sample(Q\_t)\\
      \ \ m\_new = MPIW\_moments(z, Q\_t)\ \ \ \ \ \# E-step\\
      \ \ m[t] = lambda*m\_new + (1-lambda)*m[t-1]\ \# smoothing};
  \end{tikzpicture}
  \end{center}
  \vfill
  \begin{itemize}
    \item No gradients, no learning rate: just moment matching.
    \item The exponential moving average on the last line matters: a single
      noisy moment estimate can otherwise cause pathological collapse
      (e.g.\ near-zero effective weight on all but one sample). Provably
      converges to the unbiased, zero-variance estimate as $T \to \infty$.
  \end{itemize}
\end{frame}

\begin{frame}{Results: five real hierarchical models, ELBO vs.\ time}
  \begin{center}
    \includegraphics[width=0.76\textwidth]{figures/qem_results_time.png}
  \end{center}
  \small QEM (pink) reaches the same ELBO / predictive log-likelihood as
  MP-RWS (green) and MP-VI (orange) in a fraction of the wall-clock time, on
  Bus Breakdown, MovieLens100K, Occupancy, Radon, and Covid (Heap, Bowyer \&
  Aitchison, 2025). No backprop through the whole joint per step.
\end{frame}

\begin{frame}{Also: exactly reparameterization invariant}
  \begin{itemize}
    \item \textbf{At least as accurate}: posterior first-moment MSE (vs.\
      long HMC runs) at least as good as MP-RWS, and clearly better than
      MP-VI, throughout.
    \item \alert{Reparameterization invariant, exactly}: rescale a latent by
      $\times 1/100$ (then back), and QEM's trajectory is unchanged. MP-VI
      \emph{fails to converge at all within 250 iterations} on the
      reparameterized Radon and Covid models: a real, measured failure, not
      a hypothetical one.
  \end{itemize}
\end{frame}

\section*{Wrap-up}
% =============================================================

\begin{frame}{Summary}
  \small
  \begin{itemize}
    \item TMC makes $K$-per-latent tractable by contracting along the plate
      graph.
    \item MP-RWS turns that estimator into a training algorithm, and proves
      $\hat P_{\text{MP}}$ unbiased.
    \item MPIW reframes the same estimator as a general tool for posterior
      expectations, not just RWS training.
    \item The source term trick gets samples/marginals/moments out of the
      \emph{same} computation, via autodiff, for free.
    \item QEM closes the loop: uses those moments as a gradient-free,
      reparameterization-invariant E-step, beating MP-VI and MP-RWS on both
      speed and (at worst) matching them on accuracy.
  \end{itemize}
\end{frame}
